package com.testsdlc.iims.invoiceprocessing.domain;

import java.math.BigDecimal;
import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.List;
import java.util.UUID;

import lombok.Data;
import lombok.Builder;
import lombok.AllArgsConstructor;
import lombok.NoArgsConstructor;

import javax.persistence.*;

/**
 * Core domain entity representing an Invoice in the Intelligent Invoice Management System.
 * Contains all invoice details, status tracking, and relationships to related entities.
 */
@Entity
@Table(name = "invoices")
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class Invoice {
    @Id
    @Column(name = "id")
    private UUID id;
    
    @Column(name = "invoice_number", nullable = false)
    private String invoiceNumber;
    
    @Column(name = "invoice_date", nullable = false)
    private LocalDateTime invoiceDate;
    
    @Column(name = "client_id", nullable = false)
    private UUID clientId;
    
    @Column(name = "vendor_id", nullable = false)
    private UUID vendorId;
    
    @Column(name = "template_id")
    private UUID templateId;
    
    @Column(name = "document_id", nullable = false)
    private UUID documentId;
    
    @Column(name = "status", nullable = false)
    @Enumerated(EnumType.STRING)
    private InvoiceStatus status;
    
    @Column(name = "total_amount", precision = 19, scale = 4)
    private BigDecimal totalAmount;
    
    @Column(name = "taxable_amount", precision = 19, scale = 4)
    private BigDecimal taxableAmount;
    
    @Column(name = "currency", length = 3)
    private String currency;
    
    @Column(name = "created_at", nullable = false)
    private LocalDateTime createdAt;
    
    @Column(name = "updated_at")
    private LocalDateTime updatedAt;
    
    @Column(name = "processed_at")
    private LocalDateTime processedAt;
    
    @OneToMany(cascade = CascadeType.ALL, orphanRemoval = true, mappedBy = "invoice")
    private List<InvoiceField> fields = new ArrayList<>();
    
    @OneToMany(cascade = CascadeType.ALL, orphanRemoval = true, mappedBy = "invoice")
    private List<TaxEntry> taxEntries = new ArrayList<>();
    
    @Column(name = "workflow_id")
    private UUID workflowId;
    
    @Column(name = "workflow_status")
    private String workflowStatus;
    
    @Column(name = "error_code")
    private String errorCode;
    
    @Column(name = "error_description")
    private String errorDescription;
    
    /**
     * Enum representing the possible statuses of an invoice in the processing lifecycle.
     */
    public enum InvoiceStatus {
        DRAFT, PROCESSING, VALIDATED, APPROVED, REJECTED, FINALIZED
    }
    
    /**
     * Determines if the invoice is valid based on all required fields having valid values.
     * 
     * @return true if all required fields are valid, false otherwise
     */
    public boolean isValid() {
        if (fields == null || fields.isEmpty()) {
            return false;
        }
        
        return fields.stream()
            .filter(InvoiceField::isRequired)
            .allMatch(field -> field.getValidationStatus() == InvoiceField.ValidationStatus.VALID);
    }
    
    /**
     * Calculates invoice totals based on line items and taxes.
     * Updates the totalAmount and taxableAmount fields.
     */
    public void calculateTotals() {
        if (taxEntries == null || taxEntries.isEmpty()) {
            return;
        }
        
        // Calculate taxable amount
        this.taxableAmount = taxEntries.stream()
            .map(TaxEntry::getTaxableAmount)
            .reduce(BigDecimal.ZERO, BigDecimal::add);
        
        // Calculate tax amounts
        BigDecimal totalTaxAmount = taxEntries.stream()
            .map(TaxEntry::getTaxAmount)
            .reduce(BigDecimal.ZERO, BigDecimal::add);
        
        // Calculate total amount (taxable + tax)
        this.totalAmount = this.taxableAmount.add(totalTaxAmount);
    }
    
    /**
     * Adds a field to the invoice and sets up the bidirectional relationship.
     * 
     * @param field the field to add to this invoice
     * @return the updated invoice instance
     */
    public Invoice addField(InvoiceField field) {
        if (field == null) {
            return this;
        }
        
        if (this.fields == null) {
            this.fields = new ArrayList<>();
        }
        
        field.setInvoice(this);
        this.fields.add(field);
        return this;
    }
    
    /**
     * Adds a tax entry to the invoice and sets up the bidirectional relationship.
     * 
     * @param taxEntry the tax entry to add to this invoice
     * @return the updated invoice instance
     */
    public Invoice addTaxEntry(TaxEntry taxEntry) {
        if (taxEntry == null) {
            return this;
        }
        
        if (this.taxEntries == null) {
            this.taxEntries = new ArrayList<>();
        }
        
        taxEntry.setInvoice(this);
        this.taxEntries.add(taxEntry);
        return this;
    }
}

package com.testsdlc.iims.invoiceprocessing.domain;

import java.util.UUID;
import lombok.Data;
import lombok.AllArgsConstructor;
import lombok.NoArgsConstructor;
import lombok.Builder;

import javax.persistence.*;

/**
 * Represents a field extracted from an invoice document with validation status.
 * Maps to specific template field definitions and stores the extracted values.
 */
@Entity
@Table(name = "invoice_fields")
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class InvoiceField {
    @Id
    @Column(name = "id")
    private UUID id;
    
    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "invoice_id")
    private Invoice invoice;
    
    @Column(name = "template_field_id")
    private UUID templateFieldId;
    
    @Column(name = "name", nullable = false)
    private String name;
    
    @Column(name = "value")
    private String value;
    
    @Column(name = "extraction_confidence")
    private float extractionConfidence;
    
    @Column(name = "validation_status", nullable = false)
    @Enumerated(EnumType.STRING)
    private ValidationStatus validationStatus;
    
    @Column(name = "validation_message")
    private String validationMessage;
    
    @Column(name = "required", nullable = false)
    private boolean required;
    
    /**
     * Enum defining possible validation states for an invoice field.
     */
    public enum ValidationStatus {
        VALID, INVALID, MANUAL_REVIEW
    }
    
    /**
     * Utility method to mark a field as valid.
     */
    public void markValid() {
        this.validationStatus = ValidationStatus.VALID;
        this.validationMessage = null;
    }
    
    /**
     * Utility method to mark a field as invalid with a specific message.
     * 
     * @param message the validation error message
     */
    public void markInvalid(String message) {
        this.validationStatus = ValidationStatus.INVALID;
        this.validationMessage = message;
    }
    
    /**
     * Utility method to mark a field for manual review with a specific reason.
     * 
     * @param reason the reason for requiring manual review
     */
    public void markForManualReview(String reason) {
        this.validationStatus = ValidationStatus.MANUAL_REVIEW;
        this.validationMessage = reason;
    }
}

package com.testsdlc.iims.invoiceprocessing.domain;

import java.math.BigDecimal;
import java.util.UUID;
import lombok.Data;
import lombok.AllArgsConstructor;
import lombok.NoArgsConstructor;
import lombok.Builder;

import javax.persistence.*;

/**
 * Represents a tax entry associated with an invoice.
 * Contains details about tax type, rate, and calculated amounts.
 */
@Entity
@Table(name = "tax_entries")
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class TaxEntry {
    @Id
    @Column(name = "id")
    private UUID id;
    
    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "invoice_id")
    private Invoice invoice;
    
    @Column(name = "tax_type", nullable = false)
    private String taxType;
    
    @Column(name = "tax_rate", precision = 5, scale = 2, nullable = false)
    private BigDecimal taxRate;
    
    @Column(name = "taxable_amount", precision = 19, scale = 4, nullable = false)
    private BigDecimal taxableAmount;
    
    @Column(name = "tax_amount", precision = 19, scale = 4, nullable = false)
    private BigDecimal taxAmount;
    
    /**
     * Recalculates the tax amount based on the current tax rate and taxable amount.
     */
    public void recalculateTaxAmount() {
        if (taxRate != null && taxableAmount != null) {
            this.taxAmount = taxableAmount.multiply(taxRate.divide(new BigDecimal("100")))
                .setScale(4, BigDecimal.ROUND_HALF_UP);
        }
    }
}

package com.testsdlc.iims.invoiceprocessing.repository;

import java.util.List;
import java.util.Optional;
import java.util.UUID;

import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;
import org.springframework.stereotype.Repository;

import com.testsdlc.iims.invoiceprocessing.domain.Invoice;

/**
 * Repository interface for Invoice entities providing CRUD operations and custom queries.
 */
@Repository
public interface InvoiceRepository extends JpaRepository<Invoice, UUID> {
    
    /**
     * Find all invoices with a specific status.
     * 
     * @param status the invoice status to filter by
     * @return list of matching invoices
     */
    List<Invoice> findByStatus(Invoice.InvoiceStatus status);
    
    /**
     * Find all invoices for a specific client.
     * 
     * @param clientId the client's UUID
     * @return list of client's invoices
     */
    List<Invoice> findByClientId(UUID clientId);
    
    /**
     * Find potential duplicate invoices based on invoice number and vendor.
     * 
     * @param invoiceNumber the invoice number to check
     * @param vendorId the vendor's UUID
     * @return list of potential duplicate invoices
     */
    @Query("SELECT i FROM Invoice i WHERE i.invoiceNumber = :invoiceNumber AND i.vendorId = :vendorId")
    List<Invoice> findPotentialDuplicates(
        @Param("invoiceNumber") String invoiceNumber, 
        @Param("vendorId") UUID vendorId);
    
    /**
     * Find an invoice by its associated workflow ID.
     * 
     * @param workflowId the workflow's UUID
     * @return the invoice associated with the workflow if found
     */
    @Query("SELECT i FROM Invoice i WHERE i.workflowId = :workflowId")
    Optional<Invoice> findByWorkflowId(@Param("workflowId") UUID workflowId);
    
    /**
     * Find invoices created between specific dates.
     * 
     * @param clientId the client's UUID
     * @param startDate the start date (inclusive)
     * @param endDate the end date (inclusive)
     * @return list of invoices in the date range
     */
    @Query("SELECT i FROM Invoice i WHERE i.clientId = :clientId AND i.invoiceDate BETWEEN :startDate AND :endDate")
    List<Invoice> findByClientIdAndDateRange(
        @Param("clientId") UUID clientId,
        @Param("startDate") java.time.LocalDateTime startDate,
        @Param("endDate") java.time.LocalDateTime endDate);
}

package com.testsdlc.iims.invoiceprocessing.mapper;

import org.springframework.stereotype.Component;
import org.springframework.beans.factory.annotation.Autowired;

import com.testsdlc.iims.invoiceprocessing.domain.Invoice;
import com.testsdlc.iims.invoiceprocessing.domain.InvoiceField;
import com.testsdlc.iims.invoiceprocessing.domain.TaxEntry;
import com.testsdlc.iims.invoiceprocessing.dto.InvoiceDto;
import com.testsdlc.iims.invoiceprocessing.dto.InvoiceFieldDto;
import com.testsdlc.iims.invoiceprocessing.dto.TaxEntryDto;

import java.util.stream.Collectors;

/**
 * Mapper for converting between Invoice domain objects and DTOs.
 */
@Component
public class InvoiceMapper {
    
    /**
     * Converts an Invoice entity to its DTO representation.
     * 
     * @param invoice the entity to convert
     * @return the DTO representation
     */
    public InvoiceDto toDto(Invoice invoice) {
        if (invoice == null) {
            return null;
        }
        
        return InvoiceDto.builder()
            .id(invoice.getId())
            .invoiceNumber(invoice.getInvoiceNumber())
            .invoiceDate(invoice.getInvoiceDate())
            .clientId(invoice.getClientId())
            .vendorId(invoice.getVendorId())
            .templateId(invoice.getTemplateId())
            .documentId(invoice.getDocumentId())
            .status(invoice.getStatus() != null ? invoice.getStatus().name() : null)
            .totalAmount(invoice.getTotalAmount())
            .taxableAmount(invoice.getTaxableAmount())
            .currency(invoice.getCurrency())
            .createdAt(invoice.getCreatedAt())
            .updatedAt(invoice.getUpdatedAt())
            .processedAt(invoice.getProcessedAt())
            .fields(invoice.getFields() != null ? invoice.getFields().stream()
                .map(this::toFieldDto)
                .collect(Collectors.toList()) : null)
            .taxEntries(invoice.getTaxEntries() != null ? invoice.getTaxEntries().stream()
                .map(this::toTaxEntryDto)
                .collect(Collectors.toList()) : null)
            .workflowId(invoice.getWorkflowId())
            .workflowStatus(invoice.getWorkflowStatus())
            .errorCode(invoice.getErrorCode())
            .errorDescription(invoice.getErrorDescription())
            .build();
    }
